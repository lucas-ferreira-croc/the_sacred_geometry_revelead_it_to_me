find_program(GLSLC_EXECUTABLE glslc REQUIRED)
function(compile_vk_shader SHADER_NAME TARGET_DEPENDENCY)
    set(SHADER_PATH "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER_NAME}")
    set(SPV_OUTPUT "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER_NAME}.spv")
    
    add_custom_command(
        OUTPUT "${SPV_OUTPUT}"
        COMMAND ${GLSLC_EXECUTABLE} ${SHADER_PATH} -o ${SPV_OUTPUT}
        DEPENDS "${SHADER_PATH}"
        COMMENT "Compiling shader: ${SHADER_NAME} -> ${SHADER_NAME}.spv"
        VERBATIM
    )

    target_sources(${TARGET_DEPENDENCY} PRIVATE "${SPV_OUTPUT}")
endfunction()
